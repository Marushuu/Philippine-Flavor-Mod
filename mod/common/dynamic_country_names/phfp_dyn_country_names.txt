PHI = {
	# Standard Colony
	dynamic_country_name = {
		name = PHI
		adjective = PHI_ADJ
		use_overlord_prefix = yes

		is_main_tag_only = yes
		priority = 30

		trigger = {
			scope:actor = {
				is_subject = yes
			}
		}
	}

	# Spanish Colony
	dynamic_country_name = {
		name = phfp_dyn_c_spanish
		adjective = PHI_ADJ

		is_main_tag_only = yes
		priority = 40

		trigger = {
			scope:actor = {
				is_subject = yes
				can_increase_autonomy = yes
				overlord ?= {
					OR = {
						c:SPA = THIS
						was_formed_from = SPA
					}
				}
			}
		}
	}

	# American Colony
	dynamic_country_name = {

			scope:actor = {
				is_subject = yes

					is_subject_type = subject_type_colony
					is_subject_type = subject_type_puppet
				}
				can_increase_autonomy = yes
				overlord ?= {
					OR = {
						c:USA = THIS
						was_formed_from = USA
					}
				}
			}
		}
	}
