phfp_manila_port_button = {
    name = phfp_manila_port_button
    desc = phfp_manila_port_button_desc
    visible = {
        has_journal_entry = phfp_je_luzon_development
        NOT = {
            has_variable = phfp_manila_port_button_var
        }
    }
    possible = {
        any_scope_state = {
            state_region = s:STATE_LUZON
            owner = ROOT
            any_scope_building = {
                is_building_type = building_port
                level >= 10
            }
        }
    }
    effect = {
        set_variable = phfp_manila_port_button_var
        trigger_event = {
            id = phfp_development.4
            popup = yes
        }
    }
    ai_chance = {
        value = 10
    }
}
