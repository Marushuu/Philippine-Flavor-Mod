

phfp_integrate_tagalogs = {
    name = phfp_integrate_tagalogs
    desc = phfp_integrate_tagalogs_desc

    visible = {
        has_journal_entry = phfp_je_national_ethnogenesis_ip4
        NOT = {
            has_variable = phfp_integrated_tagalogs_var
        }
    }

    possible = {
        country_average_culture_pop_acceptance = {
            culture = cu:tagalog
            value >= acceptance_status_5
        }
    }

    effect = {
        s:STATE_LUZON = {
            remove_homeland = cu:tagalog
        }
        every_scope_pop = {
            limit = {
                culture = cu:tagalog
            }
            change_pop_culture = {
                target = cu:filipino_mestizo value = 0.25
            }
        }
        set_variable = {
            name = phfp_integrated_tagalogs_var
        }
        set_variable = {
            name = phfp_integrated_cultures_var
            value = +1
        }
    }
}
