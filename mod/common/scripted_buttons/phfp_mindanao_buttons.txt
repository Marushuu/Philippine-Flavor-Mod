phfp_arable_land_button = {
	name = phfp_arable_land_button
	desc = phfp_arable_land_button_desc

	cooldown = {
		days = 1095 # 3 Years
	}

	visible = {
		has_journal_entry = phfp_je_taming_the_south
		var:arable_land_counter_var < 3
	}

	possible = {
		custom_tooltip = {
			text = phfp_arable_land_button_tt
			var:arable_land_counter_var < 3
		}
		trigger_if = {
			limit = {
				s:STATE_MINDANAO = {
					any_scope_state = {
						owner = ROOT
						has_state_trait = phfp_state_trait_liguasan_marsh
					}
				}
			}
			has_technology_researched = intensive_agriculture
			has_technology_researched = atmospheric_engine
		}
		trigger_if = {
			limit = {
				s:STATE_MINDANAO = {
					any_scope_state = {
						owner = ROOT
						has_state_trait = phfp_state_trait_cotabato_basin
					}
				}
			}
			has_technology_researched = improved_fertilizer
			has_technology_researched = rotary_valve_engine
		}
		trigger_if = {
			limit = {
				s:STATE_MINDANAO = {
					any_scope_state = {
						owner = ROOT
						has_state_trait = phfp_state_trait_cotabato_basin_upgrade
					}
				}
			}
			has_technology_researched = mechanized_farming
			has_technology_researched = combustion_engine
		}
	}

	selected = {
		custom_tooltip = {
			text = phfp_arable_land_button_complete
			var:arable_land_counter_var >= 3
		}
	}

	effect = {
		change_variable = {
			name = arable_land_counter_var
			add = 1
		}
		trigger_event = {
			id = phfp_mindanao.9
			popup = yes
		}
	}

	ai_chance = {
		value = 10
	}
}
