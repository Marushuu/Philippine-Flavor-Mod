phfp_on_game_started = {
	effect = {
		s:STATE_LUZON = {
			add_state_trait = phfp_state_trait_central_luzon_plain
			add_arable_land = 45
			change_resource_potential = {
				type = building_iron_mine
				amount = 28
			}
			change_resource_potential = {
				type = building_rubber_plantation
				amount = 21
			}
			add_resource_potential = building_gold_field
			change_resource_potential = {
				type = building_gold_field
				amount = 4
			}
			region_state:PHI = {
				create_pop = {
					# The Sangley
					culture = min
					religion = catholic
					size = 12000
				}
				add_cultural_community = cu:min
				create_building = {
					# University of Santo Tomas
					building = building_university
					level = 1
					activate_production_methods = {
						pm_scholastic_education
						pm_religious_academia
						pm_freedom_of_movement_no_effect
					}
				}
			}
		}
		s:STATE_VISAYAS = {
			add_state_trait = phfp_state_trait_sugar_islands
			change_resource_potential = {
				type = building_coal_mine
				amount = 8
			}
			change_resource_potential = {
				type = building_rubber_plantation
				amount = 16
			}
			region_state:PHI = {
				add_cultural_community = cu:min
				create_pop = {
					culture = min
					religion = catholic
					size = 2500
				}
			}
		}
		s:STATE_MINDANAO = {
			add_state_trait = phfp_state_trait_cotabato_basin
			change_resource_potential = {
				type = building_iron_mine
				amount = 24
			}
			change_resource_potential = {
				type = building_logging_camp
				amount = 14
			}
			change_resource_potential = {
				type = building_rubber_plantation
				amount = 16
			}
			add_resource_potential = building_gold_field
			change_resource_potential = {
				type = building_gold_field
				amount = 4
			}
		}
	}
}

on_game_started = {
	on_actions = {
		phfp_on_game_started
	}
}

on_monthly_pulse_country = {
	effect = {
		if = {
			limit = {
				c:PHI ?= this
				AND = {
					has_variable = phfp_luzon_developed
					has_variable = phfp_visayas_developed
					has_variable = phfp_mindanao_developed
				}
			}
			trigger_event = {
				id = phfp_development.10
				popup = yes
			}
		}
	}
}

on_become_independent = {
	effect = {
		if = {
			limit = {
				c:PHI ?= this
			}
			trigger_event = {
				id = phfp_nationalism.5
				popup = yes
			}
		}
	}
}

on_revolution_start = {
	effect = {
		if = {
			limit = {
				scope:target ?= {
					is_revolutionary = yes
					var:civil_war_type_var ?= movement_type:phfp_movement_filipino_national
				}
			}
			trigger_event = {
				id = phfp_nationalism.3
				popup = yes
			}
		}
	}
}

on_diplo_play_start = {
	effect = {
		if = {
			limit = {
				c:PHI ?= ROOT
				any_diplomatic_play = {
					is_diplomatic_play_type = dp_independence
				}
			}
			trigger_event = {
				id = phfp_nationalism.4
				popup = yes
			}
		}
	}
}
