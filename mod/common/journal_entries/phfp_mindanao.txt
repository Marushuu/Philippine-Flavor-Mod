phfp_je_subjugation_of_the_south = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	group = je_group_historical_content

	scripted_button = phfp_maguindanao_expedition_button
	scripted_button = phfp_sulu_expedition_button
	scripted_button = phfp_establish_colonies_button

	is_shown_when_inactive = {
		c:PHI ?= this
	}

	is_shown_in_lobby = {
		c:PHI ?= this
	}

	possible = {
		c:PHI ?= this
	}

	immediate = {
		s:STATE_MINDANAO = {
			every_scope_state = {
				limit = {
					owner = ROOT
				}
				add_modifier = {
					name = phfp_md_mindanao_resistance
				}
			}
			save_scope_as = mindanao_scope
		}
		c:MGD = {
			save_scope_as = maguindanao_scope
		}
		c:SUL = {
			save_scope_as = sulu_scope
		}
	}

	complete = {
		owns_entire_state_region = STATE_MINDANAO
	}

	on_complete = {
		set_variable = phfp_subjugated_the_south_var
		s:STATE_MINDANAO = {
			every_scope_state = {
				limit = {
					owner = ROOT
				}
				remove_modifier = phfp_md_mindanao_resistance
			}
		}
		add_journal_entry = {
			type = phfp_je_taming_the_south
		}
		trigger_event = {
			id = phfp_mindanao.4
			popup = yes
		}
	}

	event_outcome_completed_effect_desc = {
		header = phfp_mindanao.4.a
		effect = {
			add_modifier = {
				name = phfp_md_moro_pacification
				days = long_modifier_time
			}
		}
	}

	fail = {
		NOT = {
			has_state_in_state_region = STATE_MINDANAO
		}
	}

	on_fail = {
		trigger_event = {
			id = phfp_mindanao.5
			popup = yes
		}
	}

	event_outcome_failed_effect_desc = {
		header = phfp_mindanao.5.a
		effect = {
			add_modifier = {
				name = phfp_md_loss_of_mindanao
				days = moderately_long_modifier_time
			}
			add_radicals = {
				value = medium_radicals
			}
			ig:ig_armed_forces ?= {
				add_modifier = {
					name = phfp_md_loss_of_mindanao_approval
					days = normal_modifier_time
				}
			}
			ig:ig_landowners ?= {
				add_modifier = {
					name = phfp_md_loss_of_mindanao
					days = normal_modifier_time
				}
			}
		}
	}

	invalid = {}

	weight = 5000
	should_be_pinned_by_default_uninvolved_or_context = yes
}

phfp_je_taming_the_south = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	group = je_group_historical_content

	scripted_button = phfp_arable_land_button
	scripted_button = phfp_send_missionaries_button
	scripted_button = phfp_fund_schools_button
	scripted_button = phfp_establish_port_of_zamboanga_button

	modifiers_while_active = {
		phfp_md_moro_pacification
	}

	immediate = {
		s:STATE_MINDANAO = {
			random_scope_state = {
				limit = {
					owner = ROOT
				}
				save_scope_as = mindanao_scope
			}
		}
		set_variable = {
			name = arable_land_counter_var
			value = 0
		}
	}
	complete = {
		scope:mindanao_scope = {
			is_incorporated = yes
			state_population >= 1000000
			gdp >= 3000000
			OR = {
				state_has_building_group_levels = {
					type = bg_light_industry
					value >= 8
				}
				state_has_building_group_levels = {
					type = bg_heavy_industry
					value >= 5
				}
				state_has_building_group_levels = {
					type = bg_mining
					value >= 15
				}
			}
			OR = {
				state_has_building_group_levels = {
					type = bg_agriculture
					value >= 10
				}
				state_has_building_group_levels = {
					type = bg_plantations
					value >= 10
				}
			}
		}
		custom_tooltip = {
			text = phfp_arable_land_button_complete
			exists = var:arable_land_counter_var
			var:arable_land_counter_var >= 3
		}
		custom_tooltip = {
			text = taming_the_south_established_port_of_zamboanga_tt
			has_variable = phfp_establish_port_of_zamboanga_var
		}
		NOT = {
			any_political_movement = {
				is_insurrectionary = yes
			}
		}
	}

	on_complete = {
		trigger_event = {
			id = phfp_mindanao.10
			popup = yes
		}
	}

	event_outcome_completed_effect_desc = {
		header = phfp_mindanao.10.a
		effect = {
			add_modifier = {
				name = phfp_md_mindanao_developed
				days = long_modifier_time
			}
			ig:ig_industrialists ?= {
				add_modifier = {
					name = phfp_md_mindanao_developed_approval
					days = normal_modifier_time
				}
			}
			ig:ig_rural_folk ?= {
				add_modifier = {
					name = phfp_md_mindanao_developed_approval
					days = normal_modifier_time
				}
			}
			s:STATE_MINDANAO = {
				every_scope_state = {
					limit = {
						owner = ROOT
					}
					add_loyalists_in_state = {
						value = medium_radicals
					}
					add_modifier = {
						name = phfp_md_tamed_south
						days = very_long_modifier_time
					}
				}
			}
		}
	}

	fail = {
		NOT = {
			has_state_in_state_region = STATE_MINDANAO
		}
	}

	on_fail = {
		trigger_event = {
			id = phfp_mindanao.5
			popup = yes
		}
	}

	event_outcome_failed_effect_desc = {
		header = phfp_mindanao.5.a
		effect = {
			add_modifier = {
				name = phfp_md_loss_of_mindanao
				days = moderately_long_modifier_time
			}
			add_radicals = {
				value = medium_radicals
			}
			ig:ig_armed_forces ?= {
				add_modifier = {
					name = phfp_md_loss_of_mindanao_approval
					days = normal_modifier_time
				}
			}
			ig:ig_landowners ?= {
				add_modifier = {
					name = phfp_md_loss_of_mindanao
					days = normal_modifier_time
				}
			}
		}
	}

	invalid = {}

	weight = 5000

	should_be_pinned_by_default_uninvolved_or_context = yes
}
