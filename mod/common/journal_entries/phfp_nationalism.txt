phfp_je_propagandist_movement = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	scripted_progress_bar = phfp_propaganda_movement_bar

	scripted_button = phfp_declare_independence_button

	is_shown_when_inactive = {
		c:PHI ?= this
	}

	possible = {
		c:PHI ?= this
		has_technology_researched = nationalism
	}

	modifiers_while_active = {
		phfp_md_ilustrado_movement
	}

	immediate = {
		trigger_event = {
			id = phfp_nationalism.1
			popup = yes
		}
		set_variable = {
			name = reliance_timer_var
			value = 0
		}
	}

	custom_completion_header = phfp_je_propagandist_movement_complete_header

	complete = {
		custom_tooltip = {
			text = the_reformists_wins_tt
			var:reliance_timer_var >= 100
		}
		AND = {
			OR = {
				is_subject_type = subject_type_dominion
				is_subject_type = subject_type_protectorate
			}
			custom_tooltip = {
				text = spain_is_overlord_tt
				top_overlord = {
					c:SPA ?= this
				}
			}
			c:SPA ?= {
				relations:ROOT >= relations_threshold:amicable
			}
		}
		average_sol >= 10
		custom_tooltip = {
			text = no_insurrectionary_movement_tt
			NOT = {
				any_political_movement = {
					is_insurrectionary = yes
				}
			}
		}
	}

	on_complete = {
		trigger_event = {
			id = phfp_nationalism.2
			popup = yes
		}
	}

	custom_failure_header = phfp_je_propagandist_movement_fail_header

	fail = {
		custom_tooltip = {
			text = the_revolution_brews_tt
			OR = {
				has_variable = force_revolution_var
				any_diplomatic_play = {
					is_diplomatic_play_type = dp_independence
					initiator_is = ROOT
				}
			}
		}
	}

	on_fail = {
		trigger_event = {
			id = phfp_nationalism.3
			popup = yes
		}
	}

	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					scope:journal_entry = {
						"scripted_bar_progress(phfp_propaganda_movement_bar)" = 400
					}
				}
				set_variable = {
					name = force_revolution_var
				}
				remove_modifier = phfp_md_lib_gain
				remove_modifier = phfp_md_lib_gain_2
				remove_modifier = phfp_md_lib_gain_3
			}
			else_if = {
				limit = {
					scope:journal_entry = {
						"scripted_bar_progress(phfp_propaganda_movement_bar)" < 400
					}
					scope:journal_entry = {
						"scripted_bar_progress(phfp_propaganda_movement_bar)" >= 300
					}
				}
				if = {
					limit = {
						NOT = {
							has_modifier = phfp_md_lib_gain_3
						}
					}
					add_modifier = {
						name = phfp_md_lib_gain_3
					}
				}
				remove_modifier = phfp_md_lib_gain
				remove_modifier = phfp_md_lib_gain_2
			}
			else_if = {
				limit = {
					scope:journal_entry = {
						"scripted_bar_progress(phfp_propaganda_movement_bar)" < 300
					}
					scope:journal_entry = {
						"scripted_bar_progress(phfp_propaganda_movement_bar)" >= 200
					}
				}
				if = {
					limit = {
						NOT = {
							has_modifier = phfp_md_lib_gain_2
						}
					}
					add_modifier = {
						name = phfp_md_lib_gain_2
					}
				}
				remove_modifier = phfp_md_lib_gain
				remove_modifier = phfp_md_lib_gain_3
			}
			else_if = {
				limit = {
					scope:journal_entry = {
						"scripted_bar_progress(phfp_propaganda_movement_bar)" < 200
					}
					scope:journal_entry = {
						"scripted_bar_progress(phfp_propaganda_movement_bar)" >= 100
					}
				}
				if = {
					limit = {
						NOT = {
							has_modifier = phfp_md_lib_gain
						}
					}
					add_modifier = {
						name = phfp_md_lib_gain
					}
				}
				remove_modifier = phfp_md_lib_gain_3
				remove_modifier = phfp_md_lib_gain_2
			}
			else_if = {
				limit = {
					scope:journal_entry = {
						"scripted_bar_progress(phfp_propaganda_movement_bar)" < 100
					}
				}
				remove_modifier = phfp_md_lib_gain
				remove_modifier = phfp_md_lib_gain_2
				remove_modifier = phfp_md_lib_gain_3
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					scope:journal_entry = {
						"scripted_bar_progress(phfp_propaganda_movement_bar)" < 100
					}
				}
				change_variable = {
					name = reliance_timer_var
					add = 1
				}
			}
		}
	}

	on_yearly_pulse = {

	}

	invalid = {
		is_subject = no
	}

	on_invalid = {
		trigger_event = {
			id = phfp_nationalism.4
			popup = yes
		}
	}

	weight = 5000
	should_be_pinned_by_default_uninvolved_or_context = yes
}

phfp_je_national_ethnogenesis_ip4 = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	group = je_group_historical_content

	scripted_button = phfp_integrate_tagalogs
	scripted_button = phfp_integrate_ilocanos
	scripted_button = phfp_integrate_visayans
	scripted_button = phfp_integrate_lumads
	scripted_button = phfp_integrate_moros

	is_shown_in_lobby = {
		c:PHI ?= this
		has_dlc_feature = ip4_content
	}

	is_shown_when_inactive = {
		c:PHI ?= this
		has_dlc_feature = ip4_content
	}

	possible = {
		has_dlc_feature = ip4_content
		OR = {
			c:PHI ?= this
			was_formed_from = PHI
		}
		country_has_primary_culture = cu:filipino_mestizo
		has_technology_researched = nationalism
	}

	custom_completion_header = je_philippines_ethnogenesis_header_independence_complete

	complete = {
		country_has_primary_culture = cu:filipino_mestizo
		is_subject = no
		literacy_rate >= 0.6
		country_average_culture_pop_acceptance = {
			culture = cu:tagalog
			value >= acceptance_status_5
		}
		country_average_culture_pop_acceptance = {
			culture = cu:ilocano
			value >= acceptance_status_5
		}
		country_average_culture_pop_acceptance = {
			culture = cu:visayan
			value >= acceptance_status_5
		}
	}

	custom_on_completion_header = je_philippines_ethnogenesis_header_independence_on_complete

	on_complete = {
		trigger_event = {
			id = philippines.11
			popup = yes
		}
	}

	event_outcome_completed_effect_desc = {
		header = philippines.11.a
		effect = {
			add_primary_culture = cu:filipino
			start_national_awakening = {
				culture = cu:filipino
				months = 120
				state_region = s:STATE_LUZON
			}
			custom_tooltip = filipino_assimilation_tooltip
			every_scope_state = {
				limit = {
					OR = {
						state_region = s:STATE_LUZON
						state_region = s:STATE_VISAYAS
						state_region = s:STATE_MINDANAO
					}
				}
				add_modifier = {
					name = modifier_popular_genesis
					days = long_modifier_time
				}
			}
		}
	}

	custom_failure_header = je_philippines_ethnogenesis_header_reliance_complete

	fail = {
		country_has_primary_culture = cu:filipino_mestizo
		any_overlord_or_above = {
			country_has_primary_culture = cu:spanish
		}
		literacy_rate >= 0.6
		country_average_culture_pop_acceptance = {
			culture = cu:tagalog
			value >= acceptance_status_5
		}
		country_average_culture_pop_acceptance = {
			culture = cu:ilocano
			value >= acceptance_status_5
		}
		country_average_culture_pop_acceptance = {
			culture = cu:visayan
			value >= acceptance_status_5
		}
	}

	custom_on_failure_header = je_philippines_ethnogenesis_header_reliance_on_complete

	on_fail = {
		trigger_event = {
			id = philippines.12
			popup = yes
		}
	}

	event_outcome_failed_effect_desc = {
		header = philippines.12.a
		effect = {
			add_primary_culture = cu:filipino_hispanophone
			start_national_awakening = {
				culture = cu:filipino_hispanophone
				months = 120
				state_region = s:STATE_LUZON
			}
			custom_tooltip = {
				text = filipino_hispanophone_assimilation_tooltip
			}
			every_scope_state = {
				limit = {
					OR = {
						state_region = s:STATE_LUZON
						state_region = s:STATE_VISAYAS
						state_region = s:STATE_MINDANAO
					}
				}
				add_modifier = {
					name = modifier_popular_genesis
					days = long_modifier_time
				}
			}
			add_modifier = {
				name = modifier_phi_spanish_vision
				days = long_modifier_time
			}
		}
	}

	weight = 5000

	should_be_pinned_by_default_uninvolved_or_context = yes
}
