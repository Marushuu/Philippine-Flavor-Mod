phfp_je_propagandist_movement = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	is_shown_when_inactive = {
		c:PHI ?= this
	}

	possible = {
		c:PHI ?= this
		has_technology_researched = nationalism
	}

	modifiers_while_active = {
		phfp_md_ilustrado_movement
	}

	immediate = {
		trigger_event = {
			id = phfp_nationalism.1
			popup = yes
		}
		if = {
			limit = {
				NOT = {
					any_political_movement = {
						is_political_movement_type = phfp_movement_filipino_national
					}
				}
			}
			create_political_movement = {
				type = phfp_movement_filipino_national
			}
		}
		random_political_movement = {
			limit = {
				is_political_movement_type = phfp_movement_filipino_national
			}
			save_scope_as = propaganda_movement_scope
		}
	}

	custom_completion_header = phfp_je_propagandist_movement_complete_header

	complete = {
		custom_tooltip = {
			text = is_dominion_or_protectorate_tt
			OR = {
				has_diplomatic_pact = {
					who = overlord
					type = dominion
					is_initiator = no
				}
				has_diplomatic_pact = {
					who = overlord
					type = protectorate
					is_initiator = no
				}
			}
		}
		custom_tooltip = {
			text = no_insurrectionary_movement_tt
			NOT = {
				any_political_movement = {
					is_insurrectionary = yes
				}
			}
		}
		overlord ?= {
			relations:ROOT >= relations_threshold:amicable
		}
		country_has_voting_franchise = yes
		NOT = {
			has_law = law_type:law_frailocracy
		}
		has_law = law_type:law_slavery_banned
		liberty_desire <= 50
	}

	on_complete = {
		trigger_event = {
			id = phfp_nationalism.2
			popup = yes
		}
	}

	event_outcome_completed_effect_desc = {
		header = phfp_nationalism.2.a
		effect = {
			if = {
				limit = {
					has_dlc_feature = ip4_content
				}
				remove_primary_culture = cu:spanish
				add_primary_culture = cu:filipino_mestizo
			}
			else = {
				remove_primary_culture = cu:spanish
				add_primary_culture = cu:filipino_hispanophone
			}
			add_modifier = {
				name = modifier_phi_spanish_vision
				days = long_modifier_time
			}
			add_modifier = {
				name = modifier_metropole_business_networks
				days = long_modifier_time
			}
		}
	}

	event_outcome_completed_effect_desc = {
		header = phfp_nationalism.2.b
		effect = {
			add_modifier = {
				name = modifier_phi_spanish_vision
				days = long_modifier_time
			}
			add_modifier = {
				name = modifier_metropole_business_networks
				days = long_modifier_time
			}
		}
	}

	event_outcome_completed_effect_desc = {
		header = phfp_nationalism.2.c
		effect = {
			philippines_annex_effects = yes
		}
	}

	custom_failure_header = phfp_je_propagandist_movement_fail_header

	fail = {
		is_subject = no
	}

	on_fail = {
	}

	event_outcome_failed_effect_desc = {
		header = phfp_nationalism.5.a
		effect = {
			if = {
				limit = {
					has_dlc_feature = ip4_content
				}
				remove_primary_culture = cu:spanish
				add_primary_culture = cu:filipino_mestizo
			}
			else = {
				remove_primary_culture = cu:spanish
				add_primary_culture = cu:filipino
			}
			add_modifier = {
				name = modifier_phi_indepenent_vision
				days = long_modifier_time
			}
			add_modifier = {
				name = modifier_national_consolidation
				days = long_modifier_time
			}
			trigger_event = {
				id = phfp_nationalism.11
			}
		}
	}

	on_monthly_pulse = {
		events = {
		}
		random_events = {
			200 = 0
			20 = phfp_gomburza.1
		}
	}

	weight = 5000
	should_be_pinned_by_default_uninvolved_or_context = yes
}
