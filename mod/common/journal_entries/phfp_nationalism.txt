phfp_je_propagandist_movement = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	is_shown_when_inactive = {
		c:PHI ?= this
	}

	possible = {
		c:PHI ?= this
		has_technology_researched = nationalism
	}

	modifiers_while_active = {
		phfp_md_ilustrado_movement
	}

	immediate = {
		trigger_event = {
			id = phfp_nationalism.1
			popup = yes
		}
		set_variable = {
			name = phfp_ilustrado_laws_var
			value = 0
		}
		if = {
			limit = {
				NOT = {
					any_political_movement = {
						is_political_movement_type = phfp_movement_filipino_national
					}
				}
			}
			create_political_movement = {
				type = phfp_movement_filipino_national
			}
		}
		random_political_movement = {
			limit = {
				is_political_movement_type = phfp_movement_filipino_national
			}
			save_scope_as = propaganda_movement_scope
		}
	}

	custom_completion_header = phfp_je_propagandist_movement_complete_header

	complete = {
		custom_tooltip = {
			text = is_dominion_or_protectorate_tt
			OR = {
				has_diplomatic_pact = {
					who = overlord
					type = dominion
					is_initiator = no
				}
				has_diplomatic_pact = {
					who = overlord
					type = protectorate
					is_initiator = no
				}
			}
		}
		custom_tooltip = {
			text = no_insurrectionary_movement_tt
			NOT = {
				any_political_movement = {
					is_insurrectionary = yes
				}
			}
		}
		overlord ?= {
			relations:ROOT >= relations_threshold:amicable
		}
		country_has_voting_franchise = yes
		NOT = {
			has_law_or_variant = law_type:law_frailocracy
			has_law_or_variant = law_type:law_legacy_slavery
		}
		liberty_desire <= 25
	}

	on_complete = {
		trigger_event = {
			id = phfp_nationalism.2
			popup = yes
		}
		set_variable = {
			name = phfp_reform_var
		}
	}

	event_outcome_completed_effect_desc = {
		header = phfp_nationalism.2.a
		effect = {

		}
	}

	event_outcome_completed_effect_desc = {
		header = phfp_nationalism.2.b
		effect = {

		}
	}

	event_outcome_completed_effect_desc = {
		header = phfp_nationalism.2.c
		effect = {

		}
	}

	custom_failure_header = phfp_je_propagandist_movement_fail_header

	fail = {
		is_subject = no
	}

	on_fail = {
		set_variable = {
			name = phfp_revolution_var
		}
	}

	on_monthly_pulse = {
		random_events = {
			120 = 0
			10 = phfp_nationalism.7
		}
	}

	on_yearly_pulse = {

	}

	weight = 5000
	should_be_pinned_by_default_uninvolved_or_context = yes
}
