# Port of Manila
phfp_development.4 = {
    type = country_event
    placement = ROOT

    title = phfp_development.4.t
    desc = phfp_development.4.d
    flavor = phfp_development.4.f

    duration = 3

    event_image = {
        video = "ep2_active_shipyard"
    }

    immediate = {
        s:STATE_LUZON = {
            every_scope_state = {
                limit = {
                    owner = ROOT
                }
                save_scope_as = luzon_scope
            }
        }
    }

    option = {
        name = phfp_development.4.a
        default_option = yes
        scope:luzon_scope.state_region = {
            remove_state_trait = state_trait_manila_bay
            add_state_trait = phfp_state_trait_port_of_manila
        }
    }
}

phfp_development.5 = {
    type = country_event
    placement = ROOT

    title = phfp_development.5.t
    desc = phfp_development.5.d
    flavor = phfp_development.5.f

    duration = 3

    event_image = {
        video = "unspecific_trains"
    }

    immediate = {
        s:STATE_LUZON = {
            every_scope_state = {
                limit = {
                    owner = ROOT
                }
                save_scope_as = luzon_scope
            }
        }
    }

    option = {
        name = phfp_development.5.a
        default_option = yes
        scope:luzon_scope = {
            add_modifier = {
                name = phfp_md_manila_railway
            }
        }
        add_modifier = {
            name = modifier_railway_magnates
            days = moderately_long_modifier_time
        }
        if = {
            limit = {
                NOT = {
                    has_technology_researched = steel_railway_cars
                }
            }
            add_technology_progress = {
                progress = 1500
                technology = steel_frame_buildings
            }
        }
    }
}
