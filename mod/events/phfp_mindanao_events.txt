namespace = phfp_mindanao

# Conquer Maguindanao
phfp_mindanao.1 = {
	type = country_event
	placement = ROOT
	title = phfp_mindanao.1.t
	desc = phfp_mindanao.1.d
	flavor = phfp_mindanao.1.f
	duration = 3
	event_image = {
		video = "europenorthamerica_before_the_battle"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	trigger = {
	}
	immediate = {
		c:MGD = {
			save_scope_as = mgd_scope
		}
	}
	option = {
		name = phfp_mindanao.1.a
		default_option = yes
		if = {
			limit = {
				c:MGD ?= {
					is_subject_of = c:SPA
				}
			}
			c:PHI ?= {
				annex = c:MGD
			}
		}
		else = {
			create_diplomatic_play = {
				type = dp_annex_war
				target_country = c:MGD
				add_war_goal = {
					holder = C:MGD
					type = return_state
					target_state = s:STATE_MINDANAO.region_state:PHI
				}
			}
		}
	}
	option = {
		name = phfp_mindanao.1.b
		remove_variable = phfp_maguindanao_expedition_button_var
	}
}

# Encourage Colonization
phfp_mindanao.3 = {
	type = country_event
	placement = ROOT
	title = phfp_mindanao.3.t
	desc = phfp_mindanao.3.d
	flavor = phfp_mindanao.3.f
	duration = 3
	event_image = {
		video = "southamerica_aristocrats"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	immediate = {
		s:STATE_MINDANAO.region_state:PHI = {
			save_scope_as = mindanao_scope
		}
		s:STATE_VISAYAS.region_state:PHI = {
			save_scope_as = visayas_scope
		}
	}
	option = {
		name = phfp_mindanao.3.a
		default_option = yes
		scope:mindanao_scope = {
			add_modifier = {
				name = phfp_md_southern_settlement
				days = very_long_modifier_time
			}
		}
		scope:visayas_scope = {
			custom_tooltip = {
				text = phfp_mindanao.3.tt_1
				every_scope_pop = {
					limit = {
						culture = cu:visayan
					}
					move_partial_pop = {
						state = scope:mindanao_scope
						population_ratio = 0.05
					}
				}
			}
		}
		ai_chance = {
			base = 60
		}
	}
	option = {
		name = phfp_mindanao.3.b
		scope:mindanao_scope = {
			add_radicals_in_state = {
				value = large_radicals
				culture = cu:visayan
			}
			add_loyalists_in_state = {
				value = large_radicals
				culture = cu:moro
			}
			add_modifier = {
				name = phfp_md_moro_autonomy
				days = very_long_modifier_time
			}
		}
		ai_chance = {
			base = 20
		}
	}
}

	immediate = {
		s:STATE_MINDANAO = {

	option = {

		}

			add_modifier = {
				name = phfp_md_loss_of_mindanao_approval
				days = normal_modifier_time
			}
		}
		ig:ig_landowners ?= {
			add_modifier = {
				name = phfp_md_loss_of_mindanao
				days = normal_modifier_time
			}
		}
	}
}
