# Conquer Maguindanao
phfp_mindanao.1 = {
	type = country_event
	placement = ROOT
	title = phfp_mindanao.1.t
	desc = phfp_mindanao.1.d
	flavor = phfp_mindanao.1.f
	duration = 3
	event_image = {
		video = "europenorthamerica_before_the_battle"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	trigger = {
	}
	immediate = {
		c:MGD = {
			save_scope_as = mgd_scope
		}
	}
	option = {
		name = phfp_mindanao.1.a
		default_option = yes
		if = {
			limit = {
				c:MGD ?= {
					is_subject_of = c:SPA
				}
			}
			c:PHI ?= {
				annex = c:MGD
			}
		}
		else = {
			create_diplomatic_play = {
				type = dp_annex_war
				target_country = c:MGD
				add_war_goal = {
					holder = C:MGD
					type = return_state
					target_state = s:STATE_MINDANAO.region_state:PHI
				}
			}
		}
	}
	option = {
		name = phfp_mindanao.1.b
		remove_variable = phfp_maguindanao_expedition_button_var
	}
}

# Encourage Colonization
phfp_mindanao.3 = {
	type = country_event
	placement = ROOT
	title = phfp_mindanao.3.t
	desc = phfp_mindanao.3.d
	flavor = phfp_mindanao.3.f
	duration = 3
	event_image = {
		video = "asia_poor_people_moving"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	immediate = {
		s:STATE_MINDANAO.region_state:PHI = {
			save_scope_as = mindanao_scope
		}
		s:STATE_VISAYAS.region_state:PHI = {
			save_scope_as = visayas_scope
		}
	}
	option = {
		name = phfp_mindanao.3.a
		default_option = yes
		scope:mindanao_scope = {
			add_modifier = {
				name = phfp_md_southern_settlement
				days = very_long_modifier_time
			}
		}
		scope:visayas_scope = {
			custom_tooltip = {
				text = phfp_mindanao.3.tt_1
				every_scope_pop = {
					limit = {
						culture = cu:visayan
					}
					move_partial_pop = {
						state = scope:mindanao_scope
						population_ratio = 0.05
					}
				}
			}
		}
		ai_chance = {
			base = 60
		}
	}
	option = {
		name = phfp_mindanao.3.b
		scope:mindanao_scope = {
			add_radicals_in_state = {
				value = large_radicals
				culture = cu:visayan
			}
			add_loyalists_in_state = {
				value = large_radicals
				culture = cu:moro
			}
			add_modifier = {
				name = phfp_md_moro_autonomy
				days = very_long_modifier_time
			}
		}
		ai_chance = {
			base = 20
		}
	}
}

phfp_mindanao.9 = {
	type = country_event
	placement = ROOT
	title = phfp_mindanao.9.t
	desc = phfp_mindanao.9.d
	flavor = phfp_mindanao.9.f
	duration = 3
	event_image = {
		video = "ip2_india_countryside"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	immediate = {
		s:STATE_MINDANAO.region_state:PHI = {
			save_scope_as = mindanao_scope
		}
	}
	option = {
		name = phfp_mindanao.9.a
		default_option = yes
		scope:mindanao_scope = {
			if = {
				limit = {
					has_state_trait = phfp_state_trait_liguasan_marsh
				}
				state_region = {
					remove_state_trait = phfp_state_trait_liguasan_marsh
					add_state_trait = phfp_state_trait_cotabato_basin
				}
			}
			else_if = {
				limit = {
					has_state_trait = phfp_state_trait_cotabato_basin
				}
				state_region = {
					remove_state_trait = phfp_state_trait_cotabato_basin
					add_state_trait = phfp_state_trait_cotabato_basin_upgrade
				}
			}
			else_if = {
				limit = {
					has_state_trait = phfp_state_trait_cotabato_basin_upgrade
				}
				state_region = {
					remove_state_trait = phfp_state_trait_cotabato_basin_upgrade
					add_state_trait = phfp_state_trait_cotabato_basin_upgrade_2
					change_resource_potential = {
						type = building_oil_rig
						amount = 12
					}
				}
			}
			state_region = {
				add_arable_land = 15
			}
		}
		add_modifier = {
			name = phfp_md_immigration
			days = normal_modifier_time
		}
	}
}
