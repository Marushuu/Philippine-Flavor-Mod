namespace = phfp_nationalism

# Starter Event - Ilustrados
phfp_nationalism.1 = {
    type = country_event
    placement = ROOT

    title = phfp_nationalism.1.t
    desc = phfp_nationalism.1.d
    flavor = phfp_nationalism.1.f

    duration = 3

    icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

    event_image = {
        video = "southamerica_aristocrats"
    }

    trigger = {
    }

    immediate = {
        top_overlord = {
            save_scope_as = overlord_scope
        }
        set_variable = {
            name = phfp_ilustrado_var
        }
    }

    option = {
        name = phfp_nationalism.1.a
        default_option = yes
        add_modifier = {
            name = ilustrados_education_fervor_add
            days = moderately_long_modifier_time
            is_decaying = yes
        }
        ig:ig_intelligentsia ?= {
            if = {
                limit = {
                    has_ideology = ideology:ideology_liberal
                }
                remove_ideology = ideology_liberal
                add_ideology = phfp_ideology_ilustrado
            }
            set_ig_trait = ig_trait:ig_trait_westernisation
        }
    }

    option = {
        name = phfp_nationalism.1.b
        ig:ig_intelligentsia ?= {
            if = {
                limit = {
                    has_ideology = ideology:ideology_liberal
                }
                remove_ideology = ideology_liberal
                add_ideology = phfp_ideology_ilustrado
            }
            set_ig_trait = ig_trait:ig_trait_westernisation
        }
    }
}

    type = country_event
    placement = ROOT

    title = phfp_nationalism.2.t

    flavor = phfp_nationalism.2.f

    duration = 3

    icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

    event_image = {
        video = "southamerica_aristocrats"
    }

    trigger = {
    }

    immediate = {
        top_overlord = {
            save_scope_as = overlord_scope
        }
    }

    option = {
        name = phfp_nationalism.2.a
        default_option = yes
        if = {
            limit = {
                has_dlc_feature = ip4_content
            }
            remove_primary_culture = cu:spanish
            add_primary_culture = cu:filipino_mestizo
        }
        else = {
            remove_primary_culture = cu:spanish
            add_primary_culture = cu:filipino_hispanophone
        }
    }

    option = {
        name = phfp_nationalism.2.b
    }

    option = {
        name = phfp_nationalism.2.c
        trigger = {
            is_player = yes
        }
        play_as = overlord
        overlord = {
            annex = ROOT
        }
    }
}

# Revolution
phfp_nationalism.3 = {
    type = country_event
    placement = ROOT

    title = phfp_nationalism.3.t
    desc = phfp_nationalism.3.d
    flavor = phfp_nationalism.3.f

    duration = 3

    icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

    event_image = {
        video = "southamerica_aristocrats"
    }

    trigger = {
        c:PHI ?= this
        has_civil_war_from_movement_type = movement_type:phfp_movement_filipino_national
    }

    immediate = {
        random_country = {
            limit = {
                civil_war_origin_country ?= ROOT
            }
            save_scope_as = katipunero_scope
        }
    }

    option = { # Play as the Katipunan
        name = phfp_nationalism.3.a
        trigger = {
            is_player = yes
        }
        scope:katipunero_scope = {
            add_modifier = {
                name = phfp_md_revolutionary_fervor
                days = long_modifier_time
            }
        }
        play_as = scope:katipunero_scope
    }

    option = { # Play as the Counter-Revolutionaries
        name = phfp_nationalism.3.b
        default_option = yes
    }
}
