# Reformist Win
phfp_nationalism.2 = {
    type = country_event
    placement = ROOT

    title = phfp_nationalism.2.t
    desc = phfp_nationalism.2.d
    flavor = phfp_nationalism.2.f

    duration = 3

    icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

    event_image = {
        video = "southamerica_aristocrats"
    }

    trigger = {
    }

    immediate = {
        top_overlord = {
            save_scope_as = overlord_scope
        }
    }

    option = {
        name = phfp_nationalism.2.a
        default_option = yes
        if = {
            limit = {
                has_dlc_feature = ip4_content
            }
            remove_primary_culture = cu:spanish
            add_primary_culture = cu:filipino_mestizo
        }
        else = {
            remove_primary_culture = cu:spanish
            add_primary_culture = cu:filipino_hispanophone
        }
    }

    option = {
        name = phfp_nationalism.2.b
    }
}

# Trigger Revolution
phfp_nationalism.3 = {
    type = country_event
    placement = ROOT

    title = phfp_nationalism.3.t
    desc = phfp_nationalism.3.d
    flavor = phfp_nationalism.3.f

    duration = 3

    icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

    event_image = {
        video = "southamerica_aristocrats"
    }

    trigger = {
    }

    immediate = {
        add_journal_entry = {
            type = phfp_je_the_revolution
        }
        top_overlord = {
            save_scope_as = overlord_scope
        }
        if = {
            limit = {
                NOT = {
                    any_diplomatic_play = {
                        initiator_is = ROOT
                        is_diplomatic_play_type = dp_independence
                    }
                }
            }
            create_diplomatic_play = {
                type = dp_independence
                initiator = ROOT
                target_country = top_overlord
            }
        }
    }

    option = {
        name = phfp_nationalism.3.a
        default_option = yes
    }
}
