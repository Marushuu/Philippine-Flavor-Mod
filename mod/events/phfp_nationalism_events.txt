# Katipunan Founded
phfp_nationalism.6 = {
    type = country_event
    placement = root

    duration = 3

    title = phfp_nationalism.6.t
    desc = phfp_nationalism.6.d
    flavor = phfp_nationalism.6.f

    event_image = {
        video = "ip2_india_urban_scene"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    icon = "gfx/interface/icons/event_icons/event_protest.dds"

    cooldown = {
        days = never_fire_again_modifier_time
    }

    trigger = {
        any_political_movement = {
            is_political_movement_type = phfp_movement_filipino_national
            political_movement_radicalism >=  define:NPolitics|RADICALISM_PROTESTING_THRESHOLD
        }
    }

    cancellation_trigger = {
        has_revolution = yes
    }

    immediate = {
        random_political_movement = {
            limit = {
                is_political_movement_type = phfp_movement_filipino_national
            }
            save_scope_as = propaganda_movement_scope
        }
        if = {
            limit = {
                NOT = {
                    any_political_lobby = {
                        is_political_lobby_type = lobby_anti_overlord
                    }
                }
            }
            create_political_lobby = {
                type = lobby_anti_overlord
                target = top_overlord
                add_interest_group = ig:ig_petty_bourgeoisie
                add_interest_group = ig:ig_rural_folk
                lobby_formation_reason = diplomacy
            }
            set_variable = {
                name = inc_formed
                months = 3
            }
        }
        else = {
            random_political_lobby = {
                limit = {
                    is_political_lobby_type = lobby_anti_overlord
                }
                if = {
                    limit = {
                        NOT = {
                            any_lobby_member = {
                                is_interest_group_type = ig_petty_bourgeoisie
                            }
                        }
                    }
                    add_lobby_member = ig:ig_petty_bourgeoisie
                }
                if = {
                    limit = {
                        NOT = {
                            any_lobby_member = {
                                is_interest_group_type = ig_rural_folk
                            }
                        }
                    }
                    add_lobby_member = ig:ig_rural_folk
                }
            }
        }
        random_political_lobby = {
            limit = {
                is_political_lobby_type = lobby_anti_overlord
            }
            save_scope_as = relevant_lobby
        }
        if = {
            limit = {
                NOT = {
                    any_character = {
                        has_variable = andres_bonifacio_var
                    }
                }
            }
            create_character = {
                template = PHI_andres_bonifacio
                is_agitator = yes
                save_scope_as = bonifacio_scope
            }
        }
        else = {
            random_character = {
                limit = {
                    has_variable = andres_bonifacio_var
                }
                save_scope_as = bonifacio_scope
            }
        }
    }

    option = {
        name = phfp_nationalism.6.a
        default_option = yes
        scope:propaganda_movement_scope = {
            add_movement_enthusiasm_modifier = yes
        }
        add_radicals = {
            value = small_radicals
        }
    }
}
