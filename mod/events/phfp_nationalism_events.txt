# Starter Event - Ilustrados
phfp_nationalism.1 = {
    type = country_event
    placement = ROOT

    title = phfp_nationalism.1.t
    desc = phfp_nationalism.1.d
    flavor = phfp_nationalism.1.f

    duration = 3

    icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

    event_image = {
        video = "southamerica_aristocrats"
    }

    trigger = {
    }

    immediate = {
        top_overlord = {
            save_scope_as = overlord_scope
        }
        set_variable = {
            name = phfp_ilustrado_var
        }
    }

    option = {
        name = phfp_nationalism.1.a
        default_option = yes
        add_modifier = {
            name = ilustrados_education_fervor_add
            days = moderately_long_modifier_time
            is_decaying = yes
        }
        ig:ig_intelligentsia ?= {
            if = {
                limit = {
                    has_ideology = ideology:ideology_liberal
                }
                remove_ideology = ideology_liberal
                add_ideology = phfp_ideology_ilustrado
            }
            set_ig_trait = ig_trait:ig_trait_westernisation
        }
    }

    option = {
        name = phfp_nationalism.1.b
        ig:ig_intelligentsia ?= {
            if = {
                limit = {
                    has_ideology = ideology:ideology_liberal
                }
                remove_ideology = ideology_liberal
                add_ideology = phfp_ideology_ilustrado
            }
            set_ig_trait = ig_trait:ig_trait_westernisation
        }
    }
}

    type = country_event
    placement = ROOT

    title = phfp_nationalism.3.t
    desc = phfp_nationalism.3.d
    flavor = phfp_nationalism.3.f

    duration = 3

    icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

    }

    trigger = {
        # On Action
    }

    immediate = {
        random_country = {
            limit = {
                civil_war_origin_country ?= ROOT
            }
            save_scope_as = katipunero_scope
        }
    }

    option = { # Play as the Katipunan
        name = phfp_nationalism.3.a
        trigger = {
            is_player = yes
        }
        scope:katipunero_scope = {
            add_modifier = {
                name = phfp_md_revolutionary_fervor
                days = long_modifier_time
            }
            remove_primary_culture = cu:spanish
            add_primary_culture = cu:filipino_mestizo
        }
        play_as = scope:katipunero_scope
    }

    option = { # Play as the Counter-Revolutionaries
        name = phfp_nationalism.3.b
        default_option = yes
    }
}

# Katipunan Radicalism
phfp_nationalism.7 = {
    type = country_event
    placement = root

    duration = 3

    title = phfp_nationalism.7.t
    desc = phfp_nationalism.7.d
    flavor = phfp_nationalism.7.f

    event_image = {
        video = "ip2_india_urban_scene"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    icon = "gfx/interface/icons/event_icons/event_protest.dds"

    cooldown = {
        days = never_fire_again_modifier_time
    }

    trigger = {
        any_political_movement = {
            is_political_movement_type = phfp_movement_filipino_national
            political_movement_radicalism >=  define:NPolitics|RADICALISM_PROTESTING_THRESHOLD
        }
    }

    cancellation_trigger = {
        has_revolution = yes
    }

    immediate = {
        if = {
            limit = {
                NOT = {
                    any_political_lobby = {
                        is_political_lobby_type = lobby_anti_overlord
                    }
                }
            }
            create_political_lobby = {
                type = lobby_anti_overlord
                target = top_overlord
                add_interest_group = ig:ig_petty_bourgeoisie
                add_interest_group = ig:ig_rural_folk
                lobby_formation_reason = diplomacy
            }
            set_variable = {
                name = inc_formed
                months = 3
            }
        }
        else = {
            random_political_lobby = {
                limit = {
                    is_political_lobby_type = lobby_anti_overlord
                }
                if = {
                    limit = {
                        NOT = {
                            any_lobby_member = {
                                is_interest_group_type = ig_petty_bourgeoisie
                            }
                        }
                    }
                    add_lobby_member = ig:ig_petty_bourgeoisie
                }
                if = {
                    limit = {
                        NOT = {
                            any_lobby_member = {
                                is_interest_group_type = ig_rural_folk
                            }
                        }
                    }
                    add_lobby_member = ig:ig_rural_folk
                }
            }
        }
        random_political_lobby = {
            limit = {
                is_political_lobby_type = lobby_anti_overlord
            }
            save_scope_as = relevant_lobby
        }
    }

    option = {
        name = phfp_nationalism.7.a
        default_option = yes
    }
}
